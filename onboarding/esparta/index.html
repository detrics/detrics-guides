const DATASET = 'detrics_warehouse';
const TIMEZONE = 'Europe/Madrid';

const CONVERSIONS_TAB = 'Conversiones';
const CONVERSIONS_HEADER = ['Plataforma', 'Cuenta', 'ID de cuenta', 'Conversión', 'Categoría', 'Tipo', 'Detectada el', 'Conversiones (30 días)'];
const CONVERSION_TYPES = ['Lead', 'Compra', 'Otra', 'Excluir'];

function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu('Detrics')
    .addItem('Sincronizar mapa ahora', 'syncClientMapToBigQuery')

// Lo que corre el trigger diario y el menú: sincroniza y después busca conversiones nuevas.
function syncClientMapToBigQuery() {
  ensureConversionsTab();
  TABLES.forEach(syncTables);
  appendNewConversions();
}

function syncTables(cfg) {
  const key = cfg.keyColumn || 0;
  const rows = [];
  cfg.tabs.forEach((tab) => {
    const sheet = SpreadsheetApp.getActive().getSheetByName(tab);
    if (!sheet) { console.log(`Falta la pestaña "${tab}", no se sincroniza.`); return; }
    const nRows = sheet.getLastRow() - cfg.headerRows;
    if (nRows < 1) return;
    // getDisplayValues: los IDs largos quedan como texto (sin notación científica).
    sheet
      .getRange(cfg.headerRows + 1, 1, nRows, cfg.columns.length)
      .getDisplayValues()
      .filter((r) => r[key].trim() !== '')
      .filter((r) => !cfg.skipPrefix || !r[0].trim().startsWith(cfg.skipPrefix))
      .forEach((r) => rows.push(r));
  });
  if (!rows.length) { console.log(`${cfg.tabs.join(' + ')}: sin filas, no se sincroniza.`); return; }

  const selects = rows.map((r) =>
    'SELECT ' + cfg.columns.map((c, i) => `${sqlStr(r[i])} AS ${c}`).join(', ')
  );
  const target = `\`${PROJECT_ID}.${DATASET}.${cfg.table}\``;
  bqRun(`CREATE OR REPLACE TABLE ${target} AS\n${selects.join('\nUNION ALL\n')}`);
  console.log(`${rows.length} filas sincronizadas a ${target}.`);
}

// Crea la pestaña "Conversiones" la primera vez, con la lista desplegable de tipos.
function ensureConversionsTab() {
  const ss = SpreadsheetApp.getActive();
  if (ss.getSheetByName(CONVERSIONS_TAB)) return;
  const sheet = ss.insertSheet(CONVERSIONS_TAB);
  sheet.getRange(1, 1, 1, CONVERSIONS_HEADER.length)
    .setValues([CONVERSIONS_HEADER])
    .setFontWeight('bold');
  sheet.setFrozenRows(1);
  sheet.getRange('C:C').setNumberFormat('@'); // IDs como texto
  const rule = SpreadsheetApp.newDataValidation().requireValueInList(CONVERSION_TYPES, true).setAllowInvalid(false).build();
  sheet.getRange(2, 6, sheet.getMaxRows() - 1, 1).setDataValidation(rule);
  console.log(`Pestaña "${CONVERSIONS_TAB}" creada.`);
}

// Agrega a "Conversiones" las conversiones en "Sin clasificar" de los últimos 30 días que no están en la pestaña.
function appendNewConversions() {
  const ss = SpreadsheetApp.getActive();
  const sheet = ss.getSheetByName(CONVERSIONS_TAB);
  const known = new Set();
  if (sheet.getLastRow() >= 2) {
    sheet.getRange(2, 1, sheet.getLastRow() - 1, 4).getDisplayValues()
      .forEach((r) => known.add(conversionKey(r[0], r[2], r[3])));
  }

  let rows;
  try {
    rows = bqRun(`
      SELECT Plataforma, Cuenta, ID_de_cuenta, Conversion, ANY_VALUE(Categoria), CAST(ROUND(SUM(Conversiones)) AS STRING)
      FROM \`${PROJECT_ID}.${DATASET}.master_conversions_daily\`
      WHERE Tipo = 'Sin clasificar'
        AND Fecha >= DATE_SUB(CURRENT_DATE('${TIMEZONE}'), INTERVAL 30 DAY)
      GROUP BY 1, 2, 3, 4
      ORDER BY 1, 2, 4`);
  } catch (e) {
    // master_conversions_daily todavía no existe (primeras semanas): el mapa se sincroniza igual.
    console.log(`Sin búsqueda de conversiones nuevas: ${e.message}`);
    return;
  }

  const today = Utilities.formatDate(new Date(), TIMEZONE, 'yyyy-MM-dd');
  const fresh = rows
    .filter((r) => r[3] && !known.has(conversionKey(r[0], r[2], r[3])))
    .map((r) => [r[0], r[1], r[2], r[3], r[4], '', today, Number(r[5])]);
  if (!fresh.length) { console.log('Sin conversiones nuevas.'); return; }

  const start = sheet.getLastRow() + 1;
  sheet.getRange(start, 1, fresh.length, CONVERSIONS_HEADER.length)
    .setValues(fresh)
    .setBackground(NEW_ROW_COLOR);
  console.log(`${fresh.length} conversiones nuevas agregadas a "${CONVERSIONS_TAB}".`);
}

function conversionKey(plataforma, idCuenta, conversion) {
  return [plataforma, idCuenta, conversion].map((v) => String(v).trim()).join('|');
}

function installDailyTrigger() {
  ScriptApp.getProjectTriggers()
    .filter((t) => t.getHandlerFunction() === 'syncClientMapToBigQuery')
    .forEach((t) => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('syncClientMapToBigQuery').timeBased().everyDays(1).atHour(3).inTimezone(TIMEZONE).create();
}

// Celda vacía → NULL tipado como STRING, así una columna toda vacía no queda INT64.
function sqlStr(v) {
  const s = String(v).trim();
  if (s === '') return 'CAST(NULL AS STRING)';
  return "'" + s.replace(/\\/g, '\\\\').replace(/'/g, "\\'").replace(/\n/g, '\\n') + "'";
}

// Corre una query y devuelve sus filas como arrays de texto (vacío si la query no devuelve filas).
function bqRun(sql) {
  let res = BigQuery.Jobs.query({ query: sql, useLegacySql: false }, PROJECT_ID);
  const jobId = res.jobReference.jobId;
  const location = res.jobReference.location;
  while (!res.jobComplete) {
    Utilities.sleep(1000);
    res = BigQuery.Jobs.getQueryResults(PROJECT_ID, jobId, { location });
  }
  const out = [];
  let page = res;
  while (true) {
    (page.rows || []).forEach((row) => out.push(row.f.map((c) => (c.v === null ? '' : String(c.v)))));
    if (!page.pageToken) break;
    page = BigQuery.Jobs.getQueryResults(PROJECT_ID, jobId, { location, pageToken: page.pageToken });
  }
  return out;
}
