 *   "Mapa de clientes"  → test.client_map    (cuenta → cliente)

const PROJECT_ID = 'inmediata-510016';
const DATASET = 'test';

const TABLES = [
  {
    table: 'client_map',
    tab: 'Mapa de clientes',
    headerRows: 2, // fila 1 = encabezados, fila 2 = descripciones
    columns: [
      'cliente',
      'grupo_empresarial',
      'google_ads_customer_id',
      'dv360_advertiser_id',
      'facebook_ads_account_id',
      'facebook_insights_page_id',
      'instagram_business_account_id',
      'tiktok_ads_advertiser_id',
      'tiktok_organic_handle',
      'x_ads_account_id',
      'linkedin_ads_account_id',
      'ga4_property_id',
      'notas',
      'looker_link',
      'campaign_name_prefix',
    ],
  },
  {
    table: 'campaign_map',
    tab: 'Campaign mapping',
    headerRows: 1,
    keyColumn: 3, // una fila cuenta si tiene Campaign id
    columns: ['grupo', 'plataforma', 'campana', 'campaign_id', 'cuenta', 'cliente'],
  },
];

function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu('Detrics')
    .addItem('Sincronizar mapa ahora', 'syncClientMapToBigQuery')
    .addToUi();
}

function syncClientMapToBigQuery() {
  TABLES.forEach(syncTab);
}

function syncTab(cfg) {
  const sheet = SpreadsheetApp.getActive().getSheetByName(cfg.tab);
  if (!sheet) { console.log(`Falta la pestaña "${cfg.tab}", no se sincroniza.`); return; }
  const nRows = sheet.getLastRow() - cfg.headerRows;
  if (nRows < 1) { console.log(`"${cfg.tab}" vacía, no se sincroniza.`); return; }

  // getDisplayValues: los IDs largos quedan como texto (sin notación científica).
  const key = cfg.keyColumn || 0;
  const rows = sheet
    .getRange(cfg.headerRows + 1, 1, nRows, cfg.columns.length)
    .getDisplayValues()
    .filter((r) => r[key].trim() !== '');
  if (!rows.length) { console.log(`"${cfg.tab}" vacía, no se sincroniza.`); return; }

  const selects = rows.map((r) =>
    'SELECT ' + cfg.columns.map((c, i) => `${sqlStr(r[i])} AS ${c}`).join(', ')
  );
  const target = `\`${PROJECT_ID}.${DATASET}.${cfg.table}\``;
  bqRun(`CREATE OR REPLACE TABLE ${target} AS\n${selects.join('\nUNION ALL\n')}`);
  console.log(`${rows.length} filas sincronizadas a ${target}.`);
}

function installDailyTrigger() {
  ScriptApp.getProjectTriggers()
    .filter((t) => t.getHandlerFunction() === 'syncClientMapToBigQuery')
    .forEach((t) => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('syncClientMapToBigQuery').timeBased().everyDays(1).atHour(3).create();
}

// Celda vacía → NULL tipado como STRING, así una columna toda vacía no queda INT64.
function sqlStr(v) {
  const s = String(v).trim();
  if (s === '') return 'CAST(NULL AS STRING)';
  return "'" + s.replace(/\\/g, '\\\\').replace(/'/g, "\\'").replace(/\n/g, '\\n') + "'";
}

function bqRun(sql) {
  let res = BigQuery.Jobs.query({ query: sql, useLegacySql: false }, PROJECT_ID);
  const jobId = res.jobReference.jobId;
  const location = res.jobReference.location;
  while (!res.jobComplete) {
    Utilities.sleep(1000);
    res = BigQuery.Jobs.getQueryResults(PROJECT_ID, jobId, { location });
  }
}

05 · Vista unpivot cliente × plataforma Crea la vista v_client_map. Toma el sheet en wide (una columna por plataforma) y lo pasa a long (una fila por cliente × plataforma donde hay ID). Normaliza (quita guiones del customer_id de Google, hace TRIM). 
-- 05_v_client_map.sql
-- Unpivot del mapa de clientes: una fila por (cliente × plataforma) donde haya ID.
-- Normaliza los IDs para que matcheen con _detrics_account_id de cada tabla raw.
--
-- Reglas de normalización:
--   google_ads: strip "-" del customer_id del sheet ("216-201-7617" → "2162017617")
--   todos:      TRIM, NULLIF('') para descartar celdas vacías
--
-- Columnas de salida:
--   cliente                   nombre tal cual queda en el dashboard
--   grupo                     grupo empresarial
--   campaign_name_prefix      prefijo que separa clientes que comparten cuenta
--                             (ej Google 216-201-7617 → WTA Merida / Queretaro / GDL)
--                             vacío string '' = sin prefijo (cuenta no compartida)
--   platform                  token corto: meta | google_ads | tiktok | ga4 |
--                             instagram | tiktok_organic
--   platform_account_id       el ID normalizado que matchea con _detrics_account_id
--
-- Dependencia: client_map (tabla nativa que escribe apps-script/sync-client-map.gs desde el sheet).

CREATE OR REPLACE VIEW `inmediata-510016.test.v_client_map` AS
WITH cleaned AS (
  SELECT
    TRIM(cliente)                                          AS cliente,
    TRIM(grupo_empresarial)                                AS grupo,
    COALESCE(TRIM(campaign_name_prefix), '')               AS campaign_name_prefix,
    NULLIF(REGEXP_REPLACE(COALESCE(google_ads_customer_id, ''), r'-', ''), '')
                                                           AS google_ads_id,
    NULLIF(TRIM(facebook_ads_account_id), '')              AS meta_id,
    NULLIF(TRIM(tiktok_ads_advertiser_id), '')             AS tiktok_id,
    NULLIF(TRIM(ga4_property_id), '')                      AS ga4_id,
    NULLIF(TRIM(instagram_business_account_id), '')        AS ig_id,
    NULLIF(TRIM(tiktok_organic_handle), '')                AS tiktok_organic
  FROM `inmediata-510016.test.client_map`
  WHERE cliente IS NOT NULL AND TRIM(cliente) != ''
)
SELECT cliente, grupo, campaign_name_prefix,
       'google_ads'     AS platform, google_ads_id   AS platform_account_id
FROM cleaned WHERE google_ads_id IS NOT NULL
UNION ALL
SELECT cliente, grupo, campaign_name_prefix,
       'meta'           AS platform, meta_id         AS platform_account_id
FROM cleaned WHERE meta_id IS NOT NULL
UNION ALL
SELECT cliente, grupo, campaign_name_prefix,
       'tiktok'         AS platform, tiktok_id       AS platform_account_id
FROM cleaned WHERE tiktok_id IS NOT NULL
UNION ALL
SELECT cliente, grupo, campaign_name_prefix,
       'ga4'            AS platform, ga4_id          AS platform_account_id
FROM cleaned WHERE ga4_id IS NOT NULL
UNION ALL
SELECT cliente, grupo, campaign_name_prefix,
       'instagram'      AS platform, ig_id           AS platform_account_id
FROM cleaned WHERE ig_id IS NOT NULL
UNION ALL
SELECT cliente, grupo, campaign_name_prefix,
       'tiktok_organic' AS platform, tiktok_organic  AS platform_account_id
FROM cleaned WHERE tiktok_organic IS NOT NULL;

10 · UNION canónica de las 3 raw Crea v_paid_ad_daily uniendo Meta meta_ad_daily + Google google_ads_campaign_daily + TikTok tiktok_ad_daily en un solo esquema. Google va a grano campaña con las columnas de creativo en NULL. 
-- 10_v_paid_ad_daily.sql
-- UNION canónica de las 3 raw tables de ads a grano ad. Google va a grano
-- campaña con ad_* en NULL (sus Search Ads no tienen creativo comparable).
--
-- Grano: fecha × platform × account × campaña × ad (Google: ad_* = NULL)
--
-- Nombres canónicos (todas las columnas del output):
--   date, platform, platform_account_id, account_name, currency,
--   campaign_id, campaign_name,
--   adset_id, adset_name,                            -- NULL en Google
--   ad_id, ad_name,                                  -- NULL en Google
--   ad_image_url, ad_video_url, ad_preview_link,
--   ad_creative_type, ad_destination_url, ad_copy,   -- NULL en Google
--   spend, impressions, clicks, ctr, cpc, cpm,
--   purchases, purchase_value,                       -- Meta tiene; TikTok 0; Google NULL
--   messaging_conversations, cost_per_messaging,     -- Meta solo (Nissan WhatsApp)
--   conversions, conversions_value,                  -- Google solo
--   video_views_p100                                 -- TikTok solo
--
-- Notas:
--   - Detrics convierte google-ads cost_micros a unidades de moneda (no dividir /1e6).
--   - TikTok usa stat_time_day como columna de fecha; se alias a date.
--   - meta_ad_daily tiene messaging a partir del resync del 2026-10-02.
--     Antes de que termine ese resync, messaging solo tiene datos del día actual.
--
-- Dependencias: meta_ad_daily, google_ads_campaign_daily, tiktok_ad_daily.

CREATE OR REPLACE VIEW `inmediata-510016.test.v_paid_ad_daily` AS

-- ====== META (ad grain) ====================================================
SELECT
  date                                                    AS date,
  'meta'                                                  AS platform,
  _detrics_account_id                                     AS platform_account_id,
  account_name,
  account_currency                                        AS currency,
  campaign_id,
  campaign_name,
  adset_id,
  adset_name,
  ad_id,
  ad_name,
  ad_image_url,
  ad_video_url,
  ad_preview_shareable_link                               AS ad_preview_link,
  ad_creative_type,
  ad_destination_url,
  instagram_copy                                          AS ad_copy,
  spend,
  impressions,
  clicks,
  ctr,
  cpc,
  cpm,

30 · Master consolidado Crea master_marketing_daily: UNION canónica joinada contra el mapa de clientes, con la lógica de prefijo que separa las cuentas que comparten dos marcas (WTAs en Google, Dimitri y El Flaco en Meta y Google). Es la tabla que Looker lee para la vista general y las vistas por plataforma. Columnas GA4 van vacías hasta que llegan los IDs correctos. 
-- 30_master_marketing_daily.sql
-- Master a grano fecha × cliente × grupo × platform × campaña × ad.
-- Alimenta Views 1 / 2 / 3 del dashboard. Google: ad_* = NULL.
--
-- Cliente de cada fila, en este orden:
--   1) La campaña está en campaign_map (pestaña "Campaign mapping") → ese cliente.
--      Sirve para cuentas compartidas (Plaza México: Dimitri / El Flaco / eventos propios;
--      GS Sports: WTA GDL / LPGA) y para cuentas con eventos mezclados (Meta WTA GDL).
--   2) Si no está y la cuenta tiene un solo cliente en el mapa → ese cliente.
--   3) Si no → 'Sin asignar' (visible en el reporte, 99_control.sql la lista).
--   Las campañas marcadas 'Excluir' no entran. Cuentas fuera del mapa tampoco.
-- Reemplaza la regla por prefijo de nombre de campaña (2026-10-07): el match es por ID.
--
-- Columnas GA4 (ga4_sessions, …) quedan NULL hasta 20_v_ga4_by_campaign.sql.
--
-- Dependencias: v_paid_ad_daily, v_client_map, campaign_map.

CREATE OR REPLACE VIEW `inmediata-510016.test.master_marketing_daily` AS
WITH
  -- Cliente por cuenta: solo sirve si la cuenta tiene UN cliente en el mapa.
  accounts AS (
    SELECT
      platform,
      platform_account_id,
      COUNT(DISTINCT cliente) AS clients_in_account,
      ANY_VALUE(cliente)      AS cliente_unico
    FROM `inmediata-510016.test.v_client_map`
    WHERE platform IN ('meta', 'google_ads', 'tiktok')
    GROUP BY platform, platform_account_id
  ),
  -- Cliente por campaña: pestaña "Campaign mapping" del sheet (lo completa Inmediata).
  campaigns AS (
    SELECT TRIM(campaign_id) AS campaign_id, ANY_VALUE(TRIM(cliente)) AS cliente
    FROM `inmediata-510016.test.campaign_map`
    WHERE TRIM(IFNULL(campaign_id, '')) != '' AND TRIM(IFNULL(cliente, '')) != ''
    GROUP BY 1
  ),
  grupos AS (
    SELECT cliente, ANY_VALUE(grupo) AS grupo
    FROM `inmediata-510016.test.v_client_map`
    GROUP BY cliente
  ),
  assigned AS (
    SELECT
      p.*,
      COALESCE(c.cliente, IF(a.clients_in_account = 1, a.cliente_unico, 'Sin asignar')) AS cliente
    FROM `inmediata-510016.test.v_paid_ad_daily` p
    INNER JOIN accounts a
      ON  p.platform            = a.platform
      AND p.platform_account_id = a.platform_account_id
    LEFT JOIN campaigns c
      ON CAST(p.campaign_id AS STRING) = c.campaign_id
  )
SELECT
  p.date,
  p.cliente,
  g.grupo,
  p.platform,
  p.platform_account_id,
  p.account_name,
  p.currency,
  p.campaign_id,
  p.campaign_name,
  p.adset_id,
  p.adset_name,
  p.ad_id,
  p.ad_name,
  p.ad_image_url,
  p.ad_video_url,
  p.ad_preview_link,
  p.ad_creative_type,
  p.ad_destination_url,
  p.ad_copy,
  p.spend,
  p.impressions,
  p.clicks,
  p.ctr,
  p.cpc,
  p.cpm,
  p.purchases,
  p.purchase_value,
  p.messaging_conversations,
  p.cost_per_messaging,
  p.conversions,
  p.conversions_value,
  p.video_views_p100,

  -- GA4 placeholders (completar en 20_v_ga4_by_campaign.sql cuando destraben IDs)
  CAST(NULL AS FLOAT64) AS ga4_sessions,
  CAST(NULL AS FLOAT64) AS ga4_engaged_sessions,
  CAST(NULL AS FLOAT64) AS ga4_total_revenue,
  CAST(NULL AS FLOAT64) AS ga4_transactions,
  CAST(NULL AS STRING)  AS ga4_source_medium

FROM assigned p
LEFT JOIN grupos g USING (cliente)
WHERE p.cliente != 'Excluir';

40 · Master de keywords Crea master_google_keywords_daily: Google keywords × match type a nivel palabra, con cliente y grupo del mapa. Alimenta la vista de keywords del dashboard. 
-- 40_master_google_keywords_daily.sql
-- Master a grano fecha × cliente × campaña × ad_group × keyword × match_type.
-- Alimenta View 4 del dashboard (Google Keywords).
--
-- Misma asignación de cliente que master_marketing_daily: campaign_map por ID de campaña,
-- si no, el cliente único de la cuenta, si no 'Sin asignar'. 'Excluir' no entra.
--
-- Dependencias: google_ads_keyword_daily, v_client_map, campaign_map.

CREATE OR REPLACE VIEW `inmediata-510016.test.master_google_keywords_daily` AS
WITH
  -- Cliente por cuenta: solo sirve si la cuenta tiene UN cliente en el mapa.
  accounts AS (
    SELECT
      platform,
      platform_account_id,
      COUNT(DISTINCT cliente) AS clients_in_account,

    FROM `inmediata-510016.test.v_client_map`
    WHERE platform = 'google_ads'
    GROUP BY platform, platform_account_id
  ),
  -- Cliente por campaña: pestaña "Campaign mapping" del sheet (lo completa Inmediata).

99 · Chequeos de auditoría No crea nada: son 5 SELECT para correr después del deploy. Chequean que no haya cuentas en raw sin cliente, cuentas en el mapa sin data, campañas de cuentas compartidas que caigan afuera por falta de prefijo, y totales de spend por plataforma y por cliente. 
-- 99_control.sql
-- Queries de auditoría para correr después de cada deploy del master.
-- No crea objetos, es solo SELECT. Ejecutar manualmente desde la consola
-- de BigQuery o pegar en Looker como data source de debug.

-- ============================================================================
-- 1) Cuentas que están en las raw pero NO en v_client_map
--    (data cargándose a la que nadie va a poder filtrar por cliente)
-- ============================================================================
WITH raw_accounts AS (
  SELECT DISTINCT 'meta' AS platform, _detrics_account_id AS platform_account_id
  FROM `inmediata-510016.test.meta_ad_daily`
  UNION ALL
  SELECT DISTINCT 'google_ads', _detrics_account_id
  FROM `inmediata-510016.test.google_ads_campaign_daily`
  UNION ALL
  SELECT DISTINCT 'tiktok', _detrics_account_id
  FROM `inmediata-510016.test.tiktok_ad_daily`
)
SELECT r.platform, r.platform_account_id,
       'account presente en raw sin fila en v_client_map' AS issue
FROM raw_accounts r
LEFT JOIN `inmediata-510016.test.v_client_map` c
  ON r.platform = c.platform AND r.platform_account_id = c.platform_account_id
WHERE c.platform_account_id IS NULL;

-- ============================================================================
-- 2) Cuentas en v_client_map SIN data en las raw tables
--    (puede ser cuenta nueva o fuera del transfer; chequear accountIds del transfer)
-- ============================================================================
WITH raw_accounts AS (
  SELECT DISTINCT 'meta' AS platform, _detrics_account_id AS platform_account_id
  FROM `inmediata-510016.test.meta_ad_daily`
  UNION ALL
  SELECT DISTINCT 'google_ads', _detrics_account_id
  FROM `inmediata-510016.test.google_ads_campaign_daily`
  UNION ALL
  SELECT DISTINCT 'tiktok', _detrics_account_id
  FROM `inmediata-510016.test.tiktok_ad_daily`
)
SELECT c.cliente, c.platform, c.platform_account_id,
       'mapeado en v_client_map, sin filas en la raw' AS issue
FROM `inmediata-510016.test.v_client_map` c
LEFT JOIN raw_accounts r
  ON r.platform = c.platform AND r.platform_account_id = c.platform_account_id
WHERE c.platform IN ('meta', 'google_ads', 'tiktok')
  AND r.platform_account_id IS NULL;

-- ============================================================================
-- 3) Campañas 'Sin asignar' en el master
--    (cuenta compartida o con eventos mezclados y la campaña no está en la
--     pestaña "Campaign mapping", o está sin cliente). Sumarlas al sheet.
-- ============================================================================
SELECT
  platform,
  platform_account_id,
  account_name,
  campaign_id,
  campaign_name,
  MIN(date)        AS desde,
  MAX(date)        AS hasta,
  SUM(impressions) AS impressions,
  SUM(spend)       AS spend
FROM `inmediata-510016.test.master_marketing_daily`
WHERE cliente = 'Sin asignar'
GROUP BY 1, 2, 3, 4, 5
ORDER BY spend DESC;

-- ============================================================================
-- 3b) Filas de campaign_map con un cliente que no existe en el mapa de clientes
--     (typo o cliente nuevo sin fila en "Mapa de clientes"). 'Excluir' es válido.
-- ============================================================================
SELECT cm.campaign_id, cm.campana, cm.cliente
FROM `inmediata-510016.test.campaign_map` cm
LEFT JOIN (SELECT DISTINCT cliente FROM `inmediata-510016.test.v_client_map`) c
  ON TRIM(cm.cliente) = c.cliente
WHERE TRIM(IFNULL(cm.cliente, '')) NOT IN ('', 'Excluir')
  AND c.cliente IS NULL;

-- ============================================================================
-- 4) Health check general del master
-- ============================================================================
SELECT
  platform,
  COUNT(DISTINCT cliente)                                    AS clientes,
  COUNT(DISTINCT platform_account_id)                        AS cuentas,
  COUNT(DISTINCT campaign_id)                                AS campañas,
  COUNT(*)                                                   AS filas,
  MIN(date)                                                  AS desde,
  MAX(date)                                                  AS hasta,
  SUM(spend)                                                 AS spend_total,
  SUM(impressions)                                           AS impressions_total,
  SUM(clicks)                                                AS clicks_total,
  SUM(COALESCE(purchases, conversions))                      AS results_total,
  SUM(messaging_conversations)                               AS whatsapp_total
FROM `inmediata-510016.test.master_marketing_daily`
GROUP BY platform
ORDER BY spend_total DESC;

-- ============================================================================
-- 5) Health check del master de keywords
-- ============================================================================
SELECT
  cliente,
  COUNT(DISTINCT keyword_text) AS keywords_unicas,
  COUNT(DISTINCT match_type)   AS match_types,
  SUM(spend)                   AS spend_total,
  SUM(clicks)                  AS clicks_total,
  SUM(conversions)             AS conv_total
FROM `inmediata-510016.test.master_google_keywords_daily`
WHERE date BETWEEN DATE_SUB(CURRENT_DATE('America/Mexico_City'), INTERVAL 30 DAY)
              AND CURRENT_DATE('America/Mexico_City')
GROUP BY cliente
ORDER BY spend_total DESC;
