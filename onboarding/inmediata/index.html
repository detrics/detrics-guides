// Lo que corre el trigger diario y el menú: sincroniza y después busca campañas nuevas.
function syncClientMapToBigQuery() {
  ensureNewCampaignsTab();
  TABLES.forEach(syncTables);

function syncTables(cfg) {
  const key = cfg.keyColumn || 0;
  const rows = [];
  cfg.tabs.forEach((tab) => {
    const sheet = SpreadsheetApp.getActive().getSheetByName(tab);
    if (!sheet) { console.log(`Falta la pestaña "${tab}", no se sincroniza.`); return; }
    const nRows = sheet.getLastRow() - cfg.headerRows;
    if (nRows < 1) return;
    // getDisplayValues: los IDs largos quedan como texto (sin notación científica).
    sheet
      .getRange(cfg.headerRows + 1, 1, nRows, cfg.columns.length)
      .getDisplayValues()
      .filter((r) => r[key].trim() !== '')
      .forEach((r) => rows.push(r));
  });
  if (!rows.length) { console.log(`${cfg.tabs.join(' + ')}: sin filas, no se sincroniza.`); return; }

  const selects = rows.map((r) =>
    'SELECT ' + cfg.columns.map((c, i) => `${sqlStr(r[i])} AS ${c}`).join(', ')
  );
  const target = `\`${PROJECT_ID}.${DATASET}.${cfg.table}\``;
  bqRun(`CREATE OR REPLACE TABLE ${target} AS\n${selects.join('\nUNION ALL\n')}`);
  console.log(`${rows.length} filas sincronizadas a ${target}.`);
}

// Crea la pestaña "Campañas nuevas" la primera vez, con la misma lista de clientes que "Campaign mapping".
function ensureNewCampaignsTab() {
  const ss = SpreadsheetApp.getActive();
  if (ss.getSheetByName(NEW_CAMPAIGNS_TAB)) return;
  const sheet = ss.insertSheet(NEW_CAMPAIGNS_TAB);
  sheet.getRange(1, 1, 1, NEW_CAMPAIGNS_HEADER.length)
    .setValues([NEW_CAMPAIGNS_HEADER])
    .setFontWeight('bold');
  sheet.setFrozenRows(1);
  sheet.getRange('D:D').setNumberFormat('@'); // IDs como texto
  const source = ss.getSheetByName('Campaign mapping');
  const rule = source && source.getRange('F2').getDataValidation();
  if (rule) sheet.getRange(2, 6, sheet.getMaxRows() - 1, 1).setDataValidation(rule);
  console.log(`Pestaña "${NEW_CAMPAIGNS_TAB}" creada.`);
}

// Agrega a "Campañas nuevas" las campañas en "Sin asignar" que todavía no están en ninguna pestaña.
function appendNewCampaigns() {
  const ss = SpreadsheetApp.getActive();
  const known = new Set();
  ['Campaign mapping', NEW_CAMPAIGNS_TAB].forEach((tab) => {
    const sheet = ss.getSheetByName(tab);
    if (!sheet || sheet.getLastRow() < 2) return;
    sheet.getRange(2, 4, sheet.getLastRow() - 1, 1).getDisplayValues()
      .forEach((r) => { if (r[0].trim()) known.add(r[0].trim()); });
  });

  const rows = bqRun(`
    WITH grupos AS (
      SELECT platform, platform_account_id, STRING_AGG(DISTINCT grupo, ' / ') AS grupo
      FROM \`${PROJECT_ID}.${DATASET}.v_client_map\`
      GROUP BY 1, 2
    )
    SELECT
      g.grupo,
      CASE m.Plataforma WHEN 'Meta' THEN 'Meta Ads' WHEN 'TikTok' THEN 'TikTok Ads' ELSE m.Plataforma END,
      m.Campana,
      m.ID_de_campana,
      CONCAT(m.Cuenta, ' (', IF(m.Plataforma = 'Google Ads' AND LENGTH(m.ID_de_cuenta) = 10,
        CONCAT(SUBSTR(m.ID_de_cuenta, 1, 3), '-', SUBSTR(m.ID_de_cuenta, 4, 3), '-', SUBSTR(m.ID_de_cuenta, 7)),
        m.ID_de_cuenta), ')'),
      CAST(MIN(m.Fecha) AS STRING),
      CAST(ROUND(SUM(m.Inversion)) AS STRING)
    FROM \`${PROJECT_ID}.${DATASET}.master_marketing_daily\` m
    LEFT JOIN grupos g
      ON g.platform_account_id = m.ID_de_cuenta
     AND g.platform = CASE m.Plataforma WHEN 'Meta' THEN 'meta' WHEN 'Google Ads' THEN 'google_ads' WHEN 'TikTok' THEN 'tiktok' END
    WHERE m.Cliente = 'Sin asignar'
    GROUP BY 1, 2, 3, 4, 5
    ORDER BY MIN(m.Fecha)`);

  const today = Utilities.formatDate(new Date(), 'America/Mexico_City', 'yyyy-MM-dd');
  const fresh = rows
    .filter((r) => r[3] && !known.has(r[3]))
    .map((r) => [r[0] || '', r[1], r[2], r[3], r[4], '', today, r[5], Number(r[6])]);
  if (!fresh.length) { console.log('Sin campañas nuevas.'); return; }

  const sheet = ss.getSheetByName(NEW_CAMPAIGNS_TAB);
  const start = sheet.getLastRow() + 1;
  sheet.getRange(start, 1, fresh.length, NEW_CAMPAIGNS_HEADER.length)
    .setValues(fresh)
    .setBackground(NEW_ROW_COLOR);
  console.log(`${fresh.length} campañas nuevas agregadas a "${NEW_CAMPAIGNS_TAB}".`);
}

function installDailyTrigger() {
  ScriptApp.getProjectTriggers()
    .filter((t) => t.getHandlerFunction() === 'syncClientMapToBigQuery')
    .forEach((t) => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('syncClientMapToBigQuery').timeBased().everyDays(1).atHour(3).create();
}

// Celda vacía → NULL tipado como STRING, así una columna toda vacía no queda INT64.
function sqlStr(v) {
  const s = String(v).trim();
  if (s === '') return 'CAST(NULL AS STRING)';
  return "'" + s.replace(/\\/g, '\\\\').replace(/'/g, "\\'").replace(/\n/g, '\\n') + "'";
}

// Corre una query y devuelve sus filas como arrays de texto (vacío si la query no devuelve filas).
function bqRun(sql) {
  let res = BigQuery.Jobs.query({ query: sql, useLegacySql: false }, PROJECT_ID);
  const jobId = res.jobReference.jobId;
  const location = res.jobReference.location;
  while (!res.jobComplete) {
    Utilities.sleep(1000);
    res = BigQuery.Jobs.getQueryResults(PROJECT_ID, jobId, { location });
  }
  const out = [];
  let page = res;
  while (true) {
    (page.rows || []).forEach((row) => out.push(row.f.map((c) => (c.v === null ? '' : String(c.v)))));
    if (!page.pageToken) break;
    page = BigQuery.Jobs.getQueryResults(PROJECT_ID, jobId, { location, pageToken: page.pageToken });
  }
  return out;
}

-- Master a grano fecha × cliente × grupo × platform × campaña × ad.
-- Alimenta Views 1 / 2 / 3 del dashboard. Google: el grupo de anuncios va en Conjunto_de_anuncios
-- y ad_* = NULL.

-- Cliente de cada fila, en este orden:
--   1) La campaña está en campaign_map (pestaña "Campaign mapping") → ese cliente.
--      Sirve para cuentas compartidas (Plaza México: Dimitri / El Flaco / eventos propios;
--      GS Sports: WTA GDL / LPGA) y para cuentas con eventos mezclados (Meta WTA GDL).

--   Las campañas marcadas 'Excluir' no entran. Cuentas fuera del mapa tampoco.
-- Reemplaza la regla por prefijo de nombre de campaña (2026-10-07): el match es por ID.
--
-- Columnas en español con espacios y tildes (nombres flexibles de BigQuery): Looker las muestra tal cual.
-- Nombres flexibles: sin paréntesis, barras ni puntos (BigQuery no los acepta).

-- Columnas GA4 (Sesiones GA4, …) quedan NULL hasta 20_v_ga4_by_campaign.sql.

      COUNT(DISTINCT cliente) AS clients_in_account,
      ANY_VALUE(cliente)      AS cliente_unico

    GROUP BY platform, platform_account_id
  ),
  -- Cliente por campaña: pestaña "Campaign mapping" del sheet (lo completa Inmediata).
  campaigns AS (
    SELECT TRIM(campaign_id) AS campaign_id, ANY_VALUE(TRIM(cliente)) AS cliente
    FROM `inmediata-510016.test.campaign_map`
    WHERE TRIM(IFNULL(campaign_id, '')) != '' AND TRIM(IFNULL(cliente, '')) != ''
    GROUP BY 1
  ),
  grupos AS (
    SELECT cliente, ANY_VALUE(grupo) AS grupo
    FROM `inmediata-510016.test.v_client_map`
    GROUP BY cliente
  ),
  assigned AS (
    SELECT
      p.*,
      COALESCE(c.cliente, IF(a.clients_in_account = 1, a.cliente_unico, 'Sin asignar')) AS cliente
    FROM `inmediata-510016.test.v_paid_ad_daily` p
    INNER JOIN accounts a
      ON  p.platform            = a.platform
      AND p.platform_account_id = a.platform_account_id
    LEFT JOIN campaigns c
      ON CAST(p.campaign_id AS STRING) = c.campaign_id
  )

    WHEN 'VIDEO'           THEN 'YouTube'
    WHEN 'PERFORMANCE_MAX' THEN 'Performance Max'

    ELSE p.campaign_type
  END                                    AS Tipo_de_campana,
  p.adset_name                           AS Conjunto_de_anuncios,
  p.adset_id                             AS ID_de_conjunto,
  p.ad_name                              AS Anuncio,
  p.ad_id                                AS ID_de_anuncio,
  p.ad_image_url                         AS Creativo,
  p.ad_video_url                         AS Video_del_creativo,
  p.ad_preview_link                      AS Vista_previa,
  p.ad_creative_type                     AS Formato,
  p.ad_destination_url                   AS URL_de_destino,
  p.ad_copy                              AS Texto_del_anuncio,

  -- Solo métricas sumables. CTR, CPC, CPM, ROAS y costos por resultado se calculan en Looker.
  p.spend                                AS Inversion,
  p.impressions                          AS Impresiones,
  p.clicks                               AS Clics,

  p.purchase_value                       AS Valor_de_compras,

  -- Leads de todas las plataformas: Meta = conversaciones de WhatsApp; Google = acciones de
  -- conversión de categoría de lead (formularios); TikTok = pendiente (ver Diccionario).

WITH
  cuentas AS (
    SELECT platform_account_id, ANY_VALUE(cliente) AS cliente, ANY_VALUE(grupo) AS grupo
    FROM `inmediata-510016.test.v_client_map`

    SELECT * EXCEPT (rn) FROM (
      SELECT _detrics_account_id AS id,
             DATE_SUB(DATE(_detrics_synced_at, 'America/Mexico_City'), INTERVAL 1 DAY) AS fecha,
             profile_views, website_clicks, account_views, account_likes, account_comments, account_shares, account_saves,

56 · Actividad diaria por publicación Crea master_social_posts_daily: lo que sumó cada publicación cada día, para filtrar por fecha de actividad en vez de fecha de publicación. Clic para ver el SQL. 
-- 56_master_social_posts_daily.sql
-- Actividad diaria por publicación de Instagram: lo que sumó cada publicación cada día.
-- Alimenta la tabla de publicaciones del Looker cuando el filtro de fechas tiene que ser
-- "lo que pasó en el período" y no "lo que se publicó en el período" (para eso, master_social_posts).
--
-- Fuente: ig_media_snapshot, foto diaria (full_append) del total acumulado de cada publicación de los
-- últimos 3 meses (historial del transfer = 3_months: el transfer trae las publicaciones de ese rango
-- por fecha de publicación; el timeframe de la tabla no se usa). Instagram solo informa totales de por
-- vida por publicación, así que el día sale de restar la foto contra la del día anterior.
--
-- Fecha: la corrida programada es a las 4 a. m. de México, así que la foto del día D es el cierre de
-- D-1. Se toma la primera corrida de cada día (la más cercana a medianoche), igual que los seguidores.
--
-- Primer día de cada publicación:
--   - publicada después de que empezó la serie → su total entero (toda su actividad es nueva);
--   - ya existía cuando empezó la serie → vacío (no hay foto anterior para restar).
-- Una publicación que pasa los 3 meses deja de tener filas (no resta). Una baja (me gusta quitado,
-- comentario borrado) da un valor negativo ese día; sumado en un período el total es correcto.
-- Alcance no entra: el alcance de por vida no se reparte por día (contaría dos veces a la misma persona).
-- Se dejan afuera los días sin ningún cambio.
--
-- Dependencias: ig_media_snapshot, ig_media_performance (texto, link, imagen, tipo), v_client_map.

CREATE OR REPLACE VIEW `inmediata-510016.test.master_social_posts_daily` AS
WITH
  cuentas AS (
    SELECT platform_account_id, ANY_VALUE(cliente) AS cliente, ANY_VALUE(grupo) AS grupo
    FROM `inmediata-510016.test.v_client_map`
    WHERE platform = 'instagram'
    GROUP BY 1
  ),
  fotos AS (
    SELECT * EXCEPT (rn) FROM (
      SELECT _detrics_account_id AS id, username, media_id, DATE(timestamp) AS publicada,
             DATE_SUB(DATE(_detrics_synced_at, 'America/Mexico_City'), INTERVAL 1 DAY) AS fecha,
             views, likes, comments, shares, saved,
             ROW_NUMBER() OVER (PARTITION BY media_id, DATE(_detrics_synced_at, 'America/Mexico_City') ORDER BY _detrics_synced_at ASC) AS rn
      FROM `inmediata-510016.test.ig_media_snapshot`
    ) WHERE rn = 1
  ),
  inicio AS (
    SELECT id, MIN(fecha) AS inicio_serie FROM fotos GROUP BY 1
  ),
  deltas AS (
    SELECT
      f.*, i.inicio_serie,
      LAG(f.fecha) OVER w IS NULL AS primera,
      f.views    - LAG(f.views)    OVER w AS d_views,
      f.likes    - LAG(f.likes)    OVER w AS d_likes,
      f.comments - LAG(f.comments) OVER w AS d_comments,
      f.shares   - LAG(f.shares)   OVER w AS d_shares,
      f.saved    - LAG(f.saved)    OVER w AS d_saved
    FROM fotos f
    JOIN inicio i USING (id)
    WINDOW w AS (PARTITION BY f.media_id ORDER BY f.fecha)
  ),
  diario AS (
    SELECT
      id, username, media_id, publicada, fecha,
      -- Primera foto de una publicación nueva: todo su total es actividad nueva.
      IF(primera, IF(fecha > inicio_serie, views, NULL),    d_views)    AS views,
      IF(primera, IF(fecha > inicio_serie, likes, NULL),    d_likes)    AS likes,
      IF(primera, IF(fecha > inicio_serie, comments, NULL), d_comments) AS comments,
      IF(primera, IF(fecha > inicio_serie, shares, NULL),   d_shares)   AS shares,
      IF(primera, IF(fecha > inicio_serie, saved, NULL),    d_saved)    AS saved
    FROM deltas
  ),
  meta AS (
    SELECT media_id, ANY_VALUE(media_product_type) AS media_product_type, ANY_VALUE(media_type) AS media_type,
           ANY_VALUE(permalink) AS permalink, ANY_VALUE(caption) AS caption, ANY_VALUE(media_url) AS media_url
    FROM `inmediata-510016.test.ig_media_performance`
    GROUP BY 1
  )
SELECT
  d.fecha                                AS Fecha,
  DATE_TRUNC(d.fecha, WEEK(MONDAY))      AS Semana,
  DATE_TRUNC(d.fecha, MONTH)             AS Mes,
  c.cliente                              AS Cliente,
  c.grupo                                AS Grupo_empresarial,
  'Instagram'                            AS Plataforma,
  d.username                             AS Cuenta,
  d.id                                   AS ID_de_cuenta,
  d.media_id                             AS ID_de_publicacion,
  d.publicada                            AS Fecha_de_publicacion,
  CASE m.media_product_type WHEN 'REELS' THEN 'Reel' WHEN 'FEED' THEN 'Feed' WHEN 'STORY' THEN 'Historia' ELSE m.media_product_type END AS Tipo,
  CASE m.media_type WHEN 'IMAGE' THEN 'Imagen' WHEN 'VIDEO' THEN 'Video' WHEN 'CAROUSEL_ALBUM' THEN 'Carrusel' ELSE m.media_type END AS Formato,
  m.permalink                            AS Link,
  m.caption                              AS Texto,
  m.media_url                            AS Imagen,
  d.views                                AS Visualizaciones,
  d.likes                                AS Me_gusta,
  d.comments                             AS Comentarios,
  d.shares                               AS Compartidos,
  d.saved                                AS Guardados,
  IFNULL(d.likes, 0) + IFNULL(d.comments, 0) + IFNULL(d.shares, 0) + IFNULL(d.saved, 0) AS Interacciones
FROM diario d
INNER JOIN cuentas c ON c.platform_account_id = d.id
LEFT JOIN meta m USING (media_id)
WHERE IFNULL(d.views, 0) != 0 OR IFNULL(d.likes, 0) != 0 OR IFNULL(d.comments, 0) != 0
   OR IFNULL(d.shares, 0) != 0 OR IFNULL(d.saved, 0) != 0;
