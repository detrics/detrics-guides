 * Setup (una vez):
 *   1. Pegar este archivo en Extensiones › Apps Script del sheet.
 *   2. Servicios (+) › BigQuery API.
 *   3. Correr syncClientMapToBigQuery una vez (autoriza y crea la tabla).

 * Quien instala necesita poder escribir en el dataset (BigQuery Data Editor) y crear jobs

 * Para aplicar un cambio del sheet al momento: menú Detrics › Sincronizar mapa ahora.

const CONFIG = {
  projectId: 'analytics-beauty-web',
  dataset: 'detrics_raw',
  table: 'mapa_marcas',
  tab: 'Mapa de marcas',
  headerRows: 2, // fila 1 = encabezados, fila 2 = descripciones
  columns: [
    'marca',
    'meta_ads',
    'google_ads',
    'tiktok_ads',
    'pinterest_ads',
    'ga4',
    'tiendanube',
    'mercado_libre',
    'facebook_insights',
    'instagram_insights',
    'search_console',
    'merchant_center',
    'prefijo_campanas',
    'notas',
  ],
};

function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu('Detrics')
    .addItem('Sincronizar mapa ahora', 'syncClientMapToBigQuery')
    .addToUi();
}

function syncClientMapToBigQuery() {
  const sheet = SpreadsheetApp.getActive().getSheetByName(CONFIG.tab);
  const nRows = sheet.getLastRow() - CONFIG.headerRows;
  if (nRows < 1) { console.log('Mapa vacío, no se sincroniza.'); return; }

  // getDisplayValues: los IDs largos quedan como texto (sin notación científica).
  const rows = sheet
    .getRange(CONFIG.headerRows + 1, 1, nRows, CONFIG.columns.length)
    .getDisplayValues()
    .filter((r) => r[0].trim() !== '');
  if (!rows.length) { console.log('Mapa vacío, no se sincroniza.'); return; }

  const selects = rows.map((r) =>
    'SELECT ' + CONFIG.columns.map((c, i) => `${sqlStr(r[i])} AS ${c}`).join(', ')
  );
  const target = `\`${CONFIG.projectId}.${CONFIG.dataset}.${CONFIG.table}\``;
  bqRun(`CREATE OR REPLACE TABLE ${target} AS\n${selects.join('\nUNION ALL\n')}`);
  console.log(`${rows.length} filas sincronizadas a ${target}.`);
}

function installDailyTrigger() {
  ScriptApp.getProjectTriggers()
    .filter((t) => t.getHandlerFunction() === 'syncClientMapToBigQuery')
    .forEach((t) => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('syncClientMapToBigQuery').timeBased().everyDays(1).atHour(3).create();
}

// Celda vacía → NULL tipado como STRING, así una columna toda vacía no queda INT64.
function sqlStr(v) {
  const s = String(v).trim();
  if (s === '') return 'CAST(NULL AS STRING)';
  return "'" + s.replace(/\\/g, '\\\\').replace(/'/g, "\\'").replace(/\n/g, '\\n') + "'";
}

function bqRun(sql) {
  let res = BigQuery.Jobs.query({ query: sql, useLegacySql: false }, CONFIG.projectId);
  const jobId = res.jobReference.jobId;
  const location = res.jobReference.location;
  while (!res.jobComplete) {
    Utilities.sleep(1000);
    res = BigQuery.Jobs.getQueryResults(CONFIG.projectId, jobId, { location });
  }
}
