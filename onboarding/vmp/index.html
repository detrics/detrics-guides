function syncClientMapToBigQuery() {
  TABLES.forEach(syncTable);
  appendNewCampaigns();
  appendNewUtms();
  appendNewGad();
}

function syncTable(cfg) {
  const sheet = SpreadsheetApp.getActive().getSheetByName(cfg.tab);
  if (!sheet) { console.log(`Falta la pestaña "${cfg.tab}", no se sincroniza.`); return; }
  const nRows = sheet.getLastRow() - cfg.headerRows;
  if (nRows < 1) { console.log(`"${cfg.tab}": sin filas.`); return; }
  // getDisplayValues: los IDs largos quedan como texto (sin notación científica).
  const rows = sheet
    .getRange(cfg.headerRows + 1, 1, nRows, cfg.columns.length)
    .getDisplayValues()
    .filter((r) => {
      const k = r[cfg.key].trim();
      return k !== '' && (!cfg.keyPattern || cfg.keyPattern.test(k));
    });
  if (!rows.length) { console.log(`"${cfg.tab}": sin filas.`); return; }
  const selects = rows.map((r) =>
    'SELECT ' + cfg.columns.map((c, i) => `${sqlStr(r[i])} AS ${c}`).join(', ')
  );
  bqRun(`CREATE OR REPLACE TABLE ${tbl(cfg.table)} AS\n${selects.join('\nUNION ALL\n')}`);
  console.log(`"${cfg.tab}": ${rows.length} filas → ${cfg.table}.`);
}

// Campañas con inversión en los últimos 30 días que quedaron Multimarca y no están en "Campañas".
function appendNewCampaigns() {
  const rows = bqRun(`
    SELECT \`${PROJECT_ID}.${DATASET}.plataforma_nombre\`(plataforma), campaign_id,
           ARRAY_AGG(campaign_name ORDER BY fecha DESC LIMIT 1)[OFFSET(0)], account_id,
           CAST(ROUND(SUM(spend)) AS STRING)
    FROM ${tbl('v_paid_campaign_daily')}
    WHERE marca = 'Multimarca' AND fecha >= DATE_SUB(CURRENT_DATE('${TZ}'), INTERVAL ${DIAS} DAY)
    GROUP BY plataforma, campaign_id, account_id
    HAVING SUM(spend) > 0
    ORDER BY SUM(spend) DESC`);
  appendRows('Campañas', 2, rows.map((r) => ({
    key: r[1],
    values: [r[0], r[1], r[2], r[3], Number(r[4]), '', today(), 'Nueva: falta la marca.'],
  })), { textColumns: [2, 4] });
}

// utm_campaign con órdenes pagas en los últimos 30 días que no cruzan ninguna campaña.
function appendNewUtms() {
  const rows = bqRun(`
    SELECT utm_campaign, COUNT(*), CAST(ROUND(SUM(order_total)) AS STRING),
           STRING_AGG(DISTINCT IFNULL(utm_source, '(vacío)'), ', '),
           ANY_VALUE(utm_content),
           STRING_AGG(DISTINCT IF(metodo_atribucion = 'plataforma_por_fuente', plataforma_inferida, NULL), ', ')
    FROM ${tbl('orders_attributed')}
    WHERE es_venta AND utm_campaign IS NOT NULL
      AND metodo_atribucion IN ('utm_campaign_sin_match', 'plataforma_por_fuente')
      AND fecha >= DATE_SUB(CURRENT_DATE('${TZ}'), INTERVAL ${DIAS} DAY)
    GROUP BY utm_campaign
    HAVING COUNT(*) >= 2
    ORDER BY SUM(order_total) DESC`);
  appendRows('UTM sin coincidencia', 2, rows.map((r) => ({
    key: r[0].trim().toLowerCase(),
    values: [r[5] ? `Anuncio pago (${r[5]})` : 'Sin clasificar', r[0], Number(r[1]), Number(r[2]), r[3], r[4] || '',
      '', '', `Nueva: últimos ${DIAS} días.`, today()],
  })), { afterLastKeyRow: true });
}

// gad_campaignid de Google con órdenes en los últimos 30 días que no está en la cuenta conectada.
function appendNewGad() {
  const rows = bqRun(`
    SELECT o.gad_campaignid, COUNT(*), COUNTIF(a.es_venta),
           CAST(ROUND(SUM(IF(a.es_venta, a.order_total, 0))) AS STRING)
    FROM ${tbl('orders_attributed')} a
    JOIN ${tbl('v_orders')} o USING (order_id)
    WHERE a.metodo_atribucion = 'google_campaign_fuera_de_cuenta'
      AND a.fecha >= DATE_SUB(CURRENT_DATE('${TZ}'), INTERVAL ${DIAS} DAY)
    GROUP BY 1
    ORDER BY 4 DESC`);
  appendRows('Campañas con venta y no en GADS', 1, rows.map((r) => ({
    key: r[0],
    values: [r[0], Number(r[1]), Number(r[2]), Number(r[3]), '', '', '', '', `Nueva: últimos ${DIAS} días.`, today()],
  })), { afterLastKeyRow: true, keyPattern: /^\d+$/, textColumns: [1] });
}

// Agrega las filas cuya clave no está en la pestaña. keyCol es 1-based.
// afterLastKeyRow: inserta debajo de la última fila con clave (antes de los totales y las notas).
function appendRows(tabName, keyCol, items, opts) {
  opts = opts || {};
  const sheet = SpreadsheetApp.getActive().getSheetByName(tabName);
  if (!sheet) { console.log(`Falta la pestaña "${tabName}".`); return; }
  const last = sheet.getLastRow();
  const keys = last > 1 ? sheet.getRange(2, keyCol, last - 1, 1).getDisplayValues().map((r) => r[0].trim()) : [];
  const known = new Set(keys.map((k) => k.toLowerCase()));
  const fresh = items.filter((it) => it.key && !known.has(String(it.key).toLowerCase()));
  if (!fresh.length) { console.log(`"${tabName}": nada nuevo.`); return; }

  let start;
  if (opts.afterLastKeyRow) {
    let lastKeyRow = 1;
    keys.forEach((k, i) => { if (k && (!opts.keyPattern || opts.keyPattern.test(k))) lastKeyRow = i + 2; });
    sheet.insertRowsAfter(lastKeyRow, fresh.length);
    start = lastKeyRow + 1;
  } else {
    start = last + 1;
  }
  const width = fresh[0].values.length;
  (opts.textColumns || []).forEach((c) => sheet.getRange(start, c, fresh.length, 1).setNumberFormat('@'));
  sheet.getRange(start, 1, fresh.length, width)
    .setValues(fresh.map((it) => it.values))
    .setBackground(NEW_ROW_COLOR);
  console.log(`"${tabName}": ${fresh.length} filas nuevas.`);
}

function installDailyTrigger() {
  ScriptApp.getProjectTriggers()
    .filter((t) => t.getHandlerFunction() === 'syncClientMapToBigQuery')
    .forEach((t) => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('syncClientMapToBigQuery').timeBased().everyDays(1).atHour(3).create();
}

function tbl(name) {
  return `\`${PROJECT_ID}.${DATASET}.${name}\``;
}

function today() {
  return Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd');
}

// Celda vacía → NULL tipado como STRING, así una columna toda vacía no queda INT64.
function sqlStr(v) {
  const s = String(v).trim();
  if (s === '') return 'CAST(NULL AS STRING)';
  return "'" + s.replace(/\\/g, '\\\\').replace(/'/g, "\\'").replace(/\n/g, '\\n') + "'";
}

// Corre una query y devuelve sus filas como arrays de texto (vacío si no devuelve filas).
function bqRun(sql) {
  let res = BigQuery.Jobs.query({ query: sql, useLegacySql: false }, PROJECT_ID);
  const jobId = res.jobReference.jobId;
  const location = res.jobReference.location;
  while (!res.jobComplete) {
    Utilities.sleep(1000);
    res = BigQuery.Jobs.getQueryResults(PROJECT_ID, jobId, { location });
  }
  const out = [];
  let page = res;
  while (true) {
    (page.rows || []).forEach((row) => out.push(row.f.map((c) => (c.v === null ? '' : String(c.v)))));
    if (!page.pageToken) break;
    page = BigQuery.Jobs.getQueryResults(PROJECT_ID, jobId, { location, pageToken: page.pageToken });
  }
  return out;
}

Las transferencias corren con la cuenta de servicio de Detrics y no dependen de ninguna persona. Tres piezas corren con la cuenta de quien las crea: la consulta programada que arma las tablas del dashboard (paso 6.1), el Apps Script del mapa de marcas (paso 2.2) y las fuentes de datos del Looker (paso 6.2). La consulta programada hoy corre con support@detrics.io: si esa cuenta pierde el acceso al proyecto, las tablas master_* dejan de actualizarse y el dashboard queda congelado en el último día. Pasar la consulta programada a la cuenta de Álvaro Es la consulta "VMP · orders_attributed": BigQuery la ejecuta sola todos los días a las 08:00 (hora de Argentina), cuando ya terminaron las transferencias, y reconstruye orders_attributed y las tablas master_* en detrics_raw. No se crea de nuevo: se le cambia la cuenta con la que corre.

    1. Verificar que la cuenta de Álvaro tenga el rol Propietario o BigQuery Admin en analytics-beauty-web (IAM, igual que en el paso 3.3). Cambiar la cuenta de una consulta programada pide el permiso bigquery.transfers.update, que Data Editor y Job User no incluyen.
